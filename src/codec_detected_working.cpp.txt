/* =========================================================
   UART
   ========================================================= */

static void UART1_Init(void)
{
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_USART1_CLK_ENABLE();

    GPIO_InitTypeDef GPIO_InitStruct = {0};

    GPIO_InitStruct.Pin =
        GPIO_PIN_9 |
        GPIO_PIN_10;

    GPIO_InitStruct.Mode =
        GPIO_MODE_AF_PP;

    GPIO_InitStruct.Pull =
        GPIO_PULLUP;

    GPIO_InitStruct.Speed =
        GPIO_SPEED_FREQ_VERY_HIGH;

    GPIO_InitStruct.Alternate =
        GPIO_AF7_USART1;

    HAL_GPIO_Init(
        GPIOA,
        &GPIO_InitStruct
    );


    huart1.Instance =
        USART1;

    huart1.Init.BaudRate =
        115200;

    huart1.Init.WordLength =
        UART_WORDLENGTH_8B;

    huart1.Init.StopBits =
        UART_STOPBITS_1;

    huart1.Init.Parity =
        UART_PARITY_NONE;

    huart1.Init.Mode =
        UART_MODE_TX_RX;

    huart1.Init.HwFlowCtl =
        UART_HWCONTROL_NONE;

    huart1.Init.OverSampling =
        UART_OVERSAMPLING_16;

    HAL_UART_Init(
        &huart1
    );
}


static void uart_print(
    const char *text)
{
    HAL_UART_Transmit(
        &huart1,
        (uint8_t *)text,
        strlen(text),
        HAL_MAX_DELAY
    );
}

/* =========================================================
   LCD GPIO / FSMC PINS
   ========================================================= */

static void LCD_GPIO_FSMC_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};

    __HAL_RCC_GPIOD_CLK_ENABLE();
    __HAL_RCC_GPIOE_CLK_ENABLE();
    __HAL_RCC_GPIOF_CLK_ENABLE();
    __HAL_RCC_GPIOG_CLK_ENABLE();
    __HAL_RCC_FSMC_CLK_ENABLE();


    /* LCD RESET = PD3 */

    GPIO_InitStruct.Pin =
        GPIO_PIN_3;

    GPIO_InitStruct.Mode =
        GPIO_MODE_OUTPUT_PP;

    GPIO_InitStruct.Pull =
        GPIO_NOPULL;

    GPIO_InitStruct.Speed =
        GPIO_SPEED_FREQ_HIGH;

    HAL_GPIO_Init(
        GPIOD,
        &GPIO_InitStruct
    );


    /* LCD BACKLIGHT = PF9 */

    GPIO_InitStruct.Pin =
        GPIO_PIN_9;

    GPIO_InitStruct.Mode =
        GPIO_MODE_OUTPUT_PP;

    GPIO_InitStruct.Pull =
        GPIO_NOPULL;

    GPIO_InitStruct.Speed =
        GPIO_SPEED_FREQ_HIGH;

    HAL_GPIO_Init(
        GPIOF,
        &GPIO_InitStruct
    );


    /* FSMC D4-D7 */

    GPIO_InitStruct.Pin =
        GPIO_PIN_7 |
        GPIO_PIN_8 |
        GPIO_PIN_9 |
        GPIO_PIN_10;

    GPIO_InitStruct.Mode =
        GPIO_MODE_AF_PP;

    GPIO_InitStruct.Pull =
        GPIO_NOPULL;

    GPIO_InitStruct.Speed =
        GPIO_SPEED_FREQ_VERY_HIGH;

    GPIO_InitStruct.Alternate =
        GPIO_AF12_FSMC;

    HAL_GPIO_Init(
        GPIOE,
        &GPIO_InitStruct
    );


    /* FSMC A18, D0-D3, NOE, NWE */

    GPIO_InitStruct.Pin =
        GPIO_PIN_13 |
        GPIO_PIN_14 |
        GPIO_PIN_15 |
        GPIO_PIN_0 |
        GPIO_PIN_1 |
        GPIO_PIN_4 |
        GPIO_PIN_5;

    GPIO_InitStruct.Mode =
        GPIO_MODE_AF_PP;

    GPIO_InitStruct.Pull =
        GPIO_NOPULL;

    GPIO_InitStruct.Speed =
        GPIO_SPEED_FREQ_VERY_HIGH;

    GPIO_InitStruct.Alternate =
        GPIO_AF12_FSMC;

    HAL_GPIO_Init(
        GPIOD,
        &GPIO_InitStruct
    );


    /* FSMC NE3 = PG10 */

    GPIO_InitStruct.Pin =
        GPIO_PIN_10;

    GPIO_InitStruct.Mode =
        GPIO_MODE_AF_PP;

    GPIO_InitStruct.Pull =
        GPIO_NOPULL;

    GPIO_InitStruct.Speed =
        GPIO_SPEED_FREQ_VERY_HIGH;

    GPIO_InitStruct.Alternate =
        GPIO_AF12_FSMC;

    HAL_GPIO_Init(
        GPIOG,
        &GPIO_InitStruct
    );
}


/* =========================================================
   FSMC
   ========================================================= */

static void LCD_FSMC_Init(void)
{
    SRAM_HandleTypeDef hsram = {0};

    FSMC_NORSRAM_TimingTypeDef readTiming = {0};
    FSMC_NORSRAM_TimingTypeDef writeTiming = {0};


    hsram.Instance =
        FSMC_NORSRAM_DEVICE;

    hsram.Extended =
        FSMC_NORSRAM_EXTENDED_DEVICE;


    hsram.Init.NSBank =
        FSMC_NORSRAM_BANK3;

    hsram.Init.DataAddressMux =
        FSMC_DATA_ADDRESS_MUX_DISABLE;

    hsram.Init.MemoryType =
        FSMC_MEMORY_TYPE_SRAM;

    hsram.Init.MemoryDataWidth =
        FSMC_NORSRAM_MEM_BUS_WIDTH_8;

    hsram.Init.BurstAccessMode =
        FSMC_BURST_ACCESS_MODE_DISABLE;

    hsram.Init.WaitSignalPolarity =
        FSMC_WAIT_SIGNAL_POLARITY_LOW;

    hsram.Init.WrapMode =
        FSMC_WRAP_MODE_DISABLE;

    hsram.Init.WaitSignalActive =
        FSMC_WAIT_TIMING_BEFORE_WS;

    hsram.Init.WriteOperation =
        FSMC_WRITE_OPERATION_ENABLE;

    hsram.Init.WaitSignal =
        FSMC_WAIT_SIGNAL_DISABLE;

    hsram.Init.ExtendedMode =
        FSMC_EXTENDED_MODE_ENABLE;

    hsram.Init.AsynchronousWait =
        FSMC_ASYNCHRONOUS_WAIT_DISABLE;

    hsram.Init.WriteBurst =
        FSMC_WRITE_BURST_DISABLE;

    hsram.Init.PageSize =
        FSMC_PAGE_SIZE_NONE;


    /* READ TIMING */

    readTiming.AddressSetupTime = 15;
    readTiming.AddressHoldTime = 0;
    readTiming.DataSetupTime = 60;
    readTiming.BusTurnAroundDuration = 0;
    readTiming.CLKDivision = 0;
    readTiming.DataLatency = 0;
    readTiming.AccessMode =
        FSMC_ACCESS_MODE_A;


    /* WRITE TIMING */

    writeTiming.AddressSetupTime = 9;
    writeTiming.AddressHoldTime = 0;
    writeTiming.DataSetupTime = 8;
    writeTiming.BusTurnAroundDuration = 0;
    writeTiming.CLKDivision = 0;
    writeTiming.DataLatency = 0;
    writeTiming.AccessMode =
        FSMC_ACCESS_MODE_A;


    HAL_SRAM_Init(
        &hsram,
        &readTiming,
        &writeTiming
    );
}


/* =========================================================
   LCD INITIALIZATION
   ========================================================= */

static void LCD_Init(void)
{
    LCD_GPIO_FSMC_Init();


    /* RESET LCD */

    HAL_GPIO_WritePin(
        GPIOD,
        GPIO_PIN_3,
        GPIO_PIN_RESET
    );

    HAL_Delay(100);

    HAL_GPIO_WritePin(
        GPIOD,
        GPIO_PIN_3,
        GPIO_PIN_SET
    );

    HAL_Delay(100);


    LCD_FSMC_Init();

    HAL_Delay(100);


    /* Memory access */

    LCD_WriteCommand(0x36);
    LCD_WriteData8(0x00);


    /* RGB format */

    LCD_WriteCommand(0x3A);
    LCD_WriteData8(0x65);


    /* Porch */

    LCD_WriteCommand(0xB2);

    LCD_WriteData8(0x0C);
    LCD_WriteData8(0x0C);
    LCD_WriteData8(0x00);
    LCD_WriteData8(0x33);
    LCD_WriteData8(0x33);


    /* Gate */

    LCD_WriteCommand(0xB7);
    LCD_WriteData8(0x35);


    /* VCOM */

    LCD_WriteCommand(0xBB);
    LCD_WriteData8(0x37);


    /* LCM */

    LCD_WriteCommand(0xC0);
    LCD_WriteData8(0x2C);


    LCD_WriteCommand(0xC2);
    LCD_WriteData8(0x01);


    LCD_WriteCommand(0xC3);
    LCD_WriteData8(0x12);


    LCD_WriteCommand(0xC4);
    LCD_WriteData8(0x20);


    /* Frame rate */

    LCD_WriteCommand(0xC6);
    LCD_WriteData8(0x0F);


    /* Power */

    LCD_WriteCommand(0xD0);
    LCD_WriteData8(0xA4);
    LCD_WriteData8(0xA1);


    /* Positive gamma */

    LCD_WriteCommand(0xE0);

    const uint8_t gammaPositive[] =
    {
        0xD0, 0x04, 0x0D, 0x11,
        0x13, 0x2B, 0x3F, 0x54,
        0x4C, 0x18, 0x0D, 0x0B,
        0x1F, 0x23
    };

    for (uint8_t value : gammaPositive)
    {
        LCD_WriteData8(value);
    }


    /* Negative gamma */

    LCD_WriteCommand(0xE1);

    const uint8_t gammaNegative[] =
    {
        0xD0, 0x04, 0x0C, 0x11,
        0x13, 0x2C, 0x3F, 0x44,
        0x51, 0x2F, 0x1F, 0x1F,
        0x20, 0x23
    };

    for (uint8_t value : gammaNegative)
    {
        LCD_WriteData8(value);
    }


    LCD_WriteCommand(0x21);

    LCD_WriteCommand(0x35);
    LCD_WriteData8(0x00);


    /* Sleep out */

    LCD_WriteCommand(0x11);

    HAL_Delay(120);


    /* Display ON */

    LCD_WriteCommand(0x29);


    /* Backlight ON */

    HAL_GPIO_WritePin(
        GPIOF,
        GPIO_PIN_9,
        GPIO_PIN_SET
    );

    HAL_Delay(50);
}


/* =========================================================
   LCD DRAWING
   ========================================================= */

static void LCD_SetArea(
    uint16_t x1,
    uint16_t y1,
    uint16_t x2,
    uint16_t y2)
{
    LCD_WriteCommand(0x2A);

    LCD_WriteData8(x1 >> 8);
    LCD_WriteData8(x1 & 0xFF);

    LCD_WriteData8(x2 >> 8);
    LCD_WriteData8(x2 & 0xFF);


    LCD_WriteCommand(0x2B);

    LCD_WriteData8(y1 >> 8);
    LCD_WriteData8(y1 & 0xFF);

    LCD_WriteData8(y2 >> 8);
    LCD_WriteData8(y2 & 0xFF);


    LCD_WriteCommand(0x2C);
}


static void LCD_Fill(
    uint16_t color)
{
    LCD_SetArea(
        0,
        0,
        239,
        239
    );

    for (
        uint32_t i = 0;
        i < 240UL * 240UL;
        i++
    )
    {
        LCD_WriteColor(color);
    }
}


static void LCD_FillRect(
    uint16_t x,
    uint16_t y,
    uint16_t width,
    uint16_t height,
    uint16_t color)
{
    LCD_SetArea(
        x,
        y,
        x + width - 1,
        y + height - 1
    );

    for (
        uint32_t i = 0;
        i < (uint32_t)width * height;
        i++
    )
    {
        LCD_WriteColor(color);
    }
}

static const uint8_t *GetGlyph(
    char c)
{
    if (c == ' ')
    {
        return nullptr;
    }

    for (
        uint32_t i = 0;
        i < sizeof(FONT_CHARS) - 1;
        i++
    )
    {
        if (FONT_CHARS[i] == c)
        {
            return FONT_5X7[i];
        }
    }

    return nullptr;
}


static void LCD_DrawChar(
    uint16_t x,
    uint16_t y,
    char c,
    uint16_t color,
    uint8_t scale)
{
    const uint8_t *glyph =
        GetGlyph(c);

    if (glyph == nullptr)
    {
        return;
    }

    for (
        uint8_t column = 0;
        column < 5;
        column++
    )
    {
        for (
            uint8_t row = 0;
            row < 7;
            row++
        )
        {
            if (
                glyph[column] &
                (1 << row)
            )
            {
                LCD_FillRect(
                    x + column * scale,
                    y + row * scale,
                    scale,
                    scale,
                    color
                );
            }
        }
    }
}


static void LCD_DrawText(
    uint16_t x,
    uint16_t y,
    const char *text,
    uint16_t color,
    uint8_t scale)
{
    while (*text)
    {
        LCD_DrawChar(
            x,
            y,
            *text,
            color,
            scale
        );

        x +=
            6 * scale;

        text++;
    }
}


/* =========================================================
   PREPARE TEXT FOR LCD

   Lowercase -> uppercase
   unsupported chars -> spaces
   ========================================================= */

static void LCD_PrepareText(
    const char *source,
    char *destination,
    uint32_t maxLength)
{
    uint32_t i = 0;

    while (
        *source != '\0' &&
        i < maxLength - 1
    )
    {
        char c =
            *source++;

        if (
            c >= 'a' &&
            c <= 'z'
        )
        {
            c =
                c - 'a' + 'A';
        }

        if (
            (c >= 'A' && c <= 'Z') ||
            (c >= '0' && c <= '9') ||
            c == '-' ||
            c == ' '
        )
        {
            destination[i++] =
                c;
        }
        else
        {
            destination[i++] =
                ' ';
        }
    }

    destination[i] =
        '\0';
}

else
{
    uart_print(
        "AUDIO CODEC: NOT DETECTED!\r\n"
    );
}

    LCD_ShowIdle();


    uart_print("\r\n");
    uart_print("==============================\r\n");
    uart_print("     RT-SPARK MP3 PLAYER\r\n");
    uart_print("==============================\r\n");

    uart_print("\r\nPLAYING MODE\r\n");
    uart_print("LEFT     = Volume DOWN\r\n");
    uart_print("RIGHT    = Volume UP\r\n");
    uart_print("Short UP = Change song\r\n");
    uart_print("Hold UP  = Pause / Resume\r\n");

    uart_print("\r\nSONG SELECTION MODE\r\n");
    uart_print("DOWN  = bit 2\r\n");
    uart_print("LEFT  = bit 1\r\n");
    uart_print("RIGHT = bit 0\r\n");
    uart_print("UP    = Select / Confirm\r\n\r\n");


    enum PlayerState
    {
        STATE_IDLE,
        STATE_SELECTING,
        STATE_PLAYING,
        STATE_PAUSED
    };


    PlayerState state =
        STATE_IDLE;

    PlayerState returnState =
        STATE_IDLE;


    uint8_t pendingSong =
        0;

    uint8_t currentSong =
        0;


    /* Start volume at 50% */

    uint8_t volumeLevel =
        50;


    bool selectionHasCandidate =
        false;


    GPIO_PinState previousUp =
        GPIO_PIN_SET;

    GPIO_PinState previousLeft =
        GPIO_PIN_SET;

    GPIO_PinState previousRight =
        GPIO_PIN_SET;


    uint32_t upPressStart =
        0;

    uint32_t confirmationStart =
        0;


    uint8_t capturedBit2 =
        0;

    uint8_t capturedBit1 =
        0;

    uint8_t capturedBit0 =
        0;

    uint8_t capturedSong =
        0;


    char message[100];


    while (1)
    {
        GPIO_PinState up =
            HAL_GPIO_ReadPin(
                GPIOC,
                GPIO_PIN_5
            );

        GPIO_PinState down =
            HAL_GPIO_ReadPin(
                GPIOC,
                GPIO_PIN_1
            );

        GPIO_PinState left =
            HAL_GPIO_ReadPin(
                GPIOC,
                GPIO_PIN_0
            );

        GPIO_PinState right =
            HAL_GPIO_ReadPin(
                GPIOC,
                GPIO_PIN_4
            );


        /* =================================================
           VOLUME CONTROL

           LEFT  = DOWN
           RIGHT = UP

           Only while playing or paused
           ================================================= */

        if (
            state == STATE_PLAYING ||
            state == STATE_PAUSED
        )
        {
            /* -------------------------------
               LEFT = VOLUME DOWN
               ------------------------------- */

            if (
                left == GPIO_PIN_RESET &&
                previousLeft == GPIO_PIN_SET
            )
            {
                if (volumeLevel >= 10)
                {
                    volumeLevel -= 10;
                }
                else
                {
                    volumeLevel = 0;
                }


                std::snprintf(
                    message,
                    sizeof(message),
                    "Volume: %u%%\r\n",
                    volumeLevel
                );

                uart_print(
                    message
                );


                if (
                    state ==
                    STATE_PLAYING
                )
                {
                    LCD_ShowPlaying(
                        currentSong,
                        volumeLevel
                    );
                }
                else
                {
                    LCD_ShowPaused(
                        currentSong,
                        volumeLevel
                    );
                }
            }


            /* -------------------------------
               RIGHT = VOLUME UP
               ------------------------------- */

            if (
                right == GPIO_PIN_RESET &&
                previousRight == GPIO_PIN_SET
            )
            {
                if (volumeLevel <= 90)
                {
                    volumeLevel += 10;
                }
                else
                {
                    volumeLevel = 100;
                }


                std::snprintf(
                    message,
                    sizeof(message),
                    "Volume: %u%%\r\n",
                    volumeLevel
                );

                uart_print(
                    message
                );


                if (
                    state ==
                    STATE_PLAYING
                )
                {
                    LCD_ShowPlaying(
                        currentSong,
                        volumeLevel
                    );
                }
                else
                {
                    LCD_ShowPaused(
                        currentSong,
                        volumeLevel
                    );
                }
            }
        }


        /* =================================================
           UP JUST PRESSED
           ================================================= */

        if (
            up == GPIO_PIN_RESET &&
            previousUp == GPIO_PIN_SET
        )
        {
            upPressStart =
                HAL_GetTick();


            /* Capture song bits */

            capturedBit2 =
                (
                    down ==
                    GPIO_PIN_RESET
                )
                ? 1
                : 0;


            capturedBit1 =
                (
                    left ==
                    GPIO_PIN_RESET
                )
                ? 1
                : 0;


            capturedBit0 =
                (
                    right ==
                    GPIO_PIN_RESET
                )
                ? 1
                : 0;


            capturedSong =
                (capturedBit2 << 2) |
                (capturedBit1 << 1) |
                capturedBit0;
        }


        /* =================================================
           UP JUST RELEASED
           ================================================= */

        if (
            up == GPIO_PIN_SET &&
            previousUp == GPIO_PIN_RESET
        )
        {
            uint32_t pressDuration =
                HAL_GetTick() -
                upPressStart;


            /* =============================================
               LONG PRESS
               Pause / Resume
               ============================================= */

            if (
                pressDuration >= 1000 &&
                (
                    state == STATE_PLAYING ||
                    state == STATE_PAUSED
                )
            )
            {
                if (
                    state ==
                    STATE_PLAYING
                )
                {
                    state =
                        STATE_PAUSED;


                    uart_print(
                        "\r\nPLAYER PAUSED\r\n"
                    );


                    LCD_ShowPaused(
                        currentSong,
                        volumeLevel
                    );
                }
                else
                {
                    state =
                        STATE_PLAYING;


                    uart_print(
                        "\r\nPLAYER RESUMED\r\n"
                    );


                    uart_print(
                        "Now playing: "
                    );


                    UART_PrintSong(
                        currentSong
                    );


                    LCD_ShowPlaying(
                        currentSong,
                        volumeLevel
                    );
                }
            }


            /* =============================================
               SHORT PRESS
               ============================================= */

            else if (
                pressDuration < 1000
            )
            {
                /* -----------------------------------------
                   NOT SELECTING YET

                   Short UP enters change-song mode.
                   ----------------------------------------- */

                if (
                    state !=
                    STATE_SELECTING
                )
                {
                    returnState =
                        state;


                    state =
                        STATE_SELECTING;


                    selectionHasCandidate =
                        false;


                    uart_print(
                        "\r\n"
                        "SONG SELECTION MODE\r\n"
                        "Set DOWN/LEFT/RIGHT bits.\r\n"
                        "Press UP to select.\r\n"
                    );


                    LCD_ShowSelectMode();
                }


                /* -----------------------------------------
                   SELECTION MODE

                   First UP after setting bits
                   chooses candidate.
                   ----------------------------------------- */

                else if (
                    !selectionHasCandidate
                )
                {
                    pendingSong =
                        capturedSong;


                    std::snprintf(
                        message,
                        sizeof(message),
                        "\r\n"
                        "Selected %u%u%u -> SONG %u\r\n",
                        capturedBit2,
                        capturedBit1,
                        capturedBit0,
                        pendingSong + 1
                    );


                    uart_print(
                        message
                    );


                    uart_print(
                        "Song: "
                    );


                    UART_PrintSong(
                        pendingSong
                    );


                    uart_print(
                        "Press UP again within "
                        "5 seconds to confirm.\r\n"
                    );


                    LCD_ShowSelection(
                        pendingSong
                    );


                    selectionHasCandidate =
                        true;


                    confirmationStart =
                        HAL_GetTick();
                }


                /* -----------------------------------------
                   Candidate already selected.

                   Second UP confirms.
                   ----------------------------------------- */

                else
                {
                    currentSong =
                        pendingSong;


                    state =
                        STATE_PLAYING;


                    selectionHasCandidate =
                        false;


                    uart_print(
                        "\r\n"
                        "==============================\r\n"
                        "SONG CONFIRMED!\r\n"
                        "==============================\r\n"
                    );


                    uart_print(
                        "Now playing: "
                    );


                    UART_PrintSong(
                        currentSong
                    );


                    std::snprintf(
                        message,
                        sizeof(message),
                        "Volume: %u%%\r\n",
                        volumeLevel
                    );


                    uart_print(
                        message
                    );


                    LCD_ShowPlaying(
                        currentSong,
                        volumeLevel
                    );
                }
            }
        }


        /* =================================================
           5 SECOND CONFIRMATION TIMEOUT
           ================================================= */

        if (
            state ==
                STATE_SELECTING &&
            selectionHasCandidate
        )
        {
            if (
                (
                    HAL_GetTick() -
                    confirmationStart
                )
                >= 5000
            )
            {
                uart_print(
                    "\r\n"
                    "TIMEOUT!\r\n"
                    "Song selection cancelled.\r\n"
                );


                LCD_ShowTimeout();


                HAL_Delay(
                    1000
                );


                selectionHasCandidate =
                    false;


                state =
                    returnState;


                if (
                    state ==
                    STATE_PLAYING
                )
                {
                    LCD_ShowPlaying(
                        currentSong,
                        volumeLevel
                    );
                }

                else if (
                    state ==
                    STATE_PAUSED
                )
                {
                    LCD_ShowPaused(
                        currentSong,
                        volumeLevel
                    );
                }

                else
                {
                    state =
                        STATE_IDLE;

                    LCD_ShowIdle();
                }
            }
        }


        previousUp =
            up;

        previousLeft =
            left;

        previousRight =
            right;


        HAL_Delay(
            20
        );
    }
}
